// =====================================================================
// Reentrancy fund drain -- SAFE (undesirable property FALSE)
// ---------------------------------------------------------------------
// Same undesirable property as reentrancy_unsafe.c, but following the
// checks-effects-interactions pattern: the balance is updated BEFORE the
// external call, so any re-entrant withdrawal is re-checked against the
// already-decremented balance and can never drive it negative.
//
//   agent "atk" = input("atk")  -> attacker (amount + re-entry), coalition
//
// The attacker CANNOT drain below zero:
//   -atl "<atk>F{balance < 0}"   (FALSE = SAFE, desirable outcome)
// =====================================================================

int main() {
    int balance = 100;
    while (balance > 0) {
        int amt = input("atk");      // attacker's withdrawal amount
        if (amt < 1) { amt = 1; }
        if (amt <= balance) {
            balance = balance - amt;     // state update FIRST (effects)
            int reenter = input("atk");  // external call AFTER (interactions)
            // a re-entrant call re-enters the loop and is bounded by the
            // amt <= balance guard, so the balance stays non-negative
        }
    }
    while (1) {}
}
