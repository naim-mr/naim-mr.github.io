// =====================================================================
// Arbiter mutual-exclusion VIOLATION -- UNSAFE (undesirable property TRUE)
// ---------------------------------------------------------------------
// Dual of the SYNTCOMP arbiters: instead of "the arbiter can keep mutual
// exclusion" (desirable), we ask "can the environment FORCE a double
// grant" (undesirable). This buggy arbiter has no arbitration logic: it
// is obliged to grant every request in the same step, so two simultaneous
// requests produce two simultaneous grants.
//
//   agent "env" = input("env")  -> the requesting clients (coalition)
//   (no arbiter agent: the broken design leaves no choice)
//
// The environment can force a safety violation:
//   -atl "<env>F{g1 + g2 >= 2}"   (TRUE = mutual exclusion BROKEN)
// =====================================================================

int main() {
    int g1 = 0;
    int g2 = 0;
    while (1) {
        int r1 = input("env");   // request client 1
        int r2 = input("env");   // request client 2
        g1 = 0;
        g2 = 0;
        // BUG: every request is granted immediately, no arbitration
        if (r1 != 0) { g1 = 1; }
        if (r2 != 0) { g2 = 1; }
    }
}
