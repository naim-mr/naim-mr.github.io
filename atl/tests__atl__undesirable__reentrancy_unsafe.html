// =====================================================================
// Reentrancy fund drain -- UNSAFE (undesirable property TRUE)
// ---------------------------------------------------------------------
// Smart-contract-style reentrancy (CWE-841 / "the DAO" pattern): the
// external call happens BEFORE the balance is updated, so a re-entrant
// withdrawal spends the same funds twice and drives the balance negative.
//
//   agent "atk" = input("atk")  -> attacker (amount + re-entry), coalition
//
// The attacker can drain below zero (robustly reachable):
//   -atl "<atk>F{balance < 0}"   (TRUE = VULNERABLE)
// =====================================================================

int main() {
    int balance = 100;
    while (balance > 0) {
        int amt = input("atk");      // attacker's withdrawal amount
        if (amt < 1) { amt = 1; }
        if (amt <= balance) {
            // BUG: external call (which the attacker can re-enter) fires
            // BEFORE the state update -> the same amount is withdrawn twice.
            int reenter = input("atk");
            if (reenter == 1) { balance = balance - amt; }  // re-entrant call
            balance = balance - amt;                         // original call
        }
    }
    while (1) {}
}
