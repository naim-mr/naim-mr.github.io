int main() {
  int x;
  while (x != 0) {
  	if (x < 0) {
  		x = x + 2;
  		if (x < 0) {
  			x = -x;			
  		}
  	} else {
  		x = x - 2;
  		if (x > 0) {
  			x = -x;
  		}
  	}
  }
  return 0;
}
