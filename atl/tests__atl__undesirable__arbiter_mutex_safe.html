// =====================================================================
// Arbiter mutual-exclusion VIOLATION -- SAFE (undesirable property FALSE)
// ---------------------------------------------------------------------
// Same undesirable property as arbiter_mutex_unsafe.c, but here a real
// arbiter chooses whom to grant. The arbiter is ADVERSARIAL w.r.t. the
// environment's attempt to break safety, so the environment cannot force
// a double grant.
//
//   agent "env" = input("env")  -> the requesting clients (coalition)
//   agent "arb" = input("arb")  -> the arbiter (adversary: keeps safety)
//
// The environment CANNOT force a violation:
//   -atl "<env>F{g1 + g2 >= 2}"   (FALSE = mutual exclusion holds, safe)
// =====================================================================

int main() {
    int g1 = 0;
    int g2 = 0;
    while (1) {
        int r1 = input("env");    // request client 1
        int r2 = input("env");    // request client 2
        int grant = input("arb"); // arbiter: 0 none, 1 client1, 2 client2
        g1 = 0;
        g2 = 0;
        if (grant == 1) { if (r1 != 0) { g1 = 1; } }
        if (grant == 2) { if (r2 != 0) { g2 = 1; } }
    }
}
