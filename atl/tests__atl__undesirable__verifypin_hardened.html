// =====================================================================
// VerifyPIN under fault injection -- HARDENED (undesirable property FALSE)
// ---------------------------------------------------------------------
// Inspired by FISSC, robust-reachability / fault-injection benchmark.
//   "Inference of Robust Reachability Constraints", POPL 2024,
//   https://zenodo.org/records/8421879
// ---------------------------------------------------------------------
// Same auth-bypass property as verifypin_unsafe.c, but the comparison is
// DUPLICATED (a classic single-fault countermeasure): both redundant
// checks must agree before granting. A single injected fault can only
// corrupt one of them, so the bypass is NO LONGER robustly reachable.
//
//   agent "atk" = input("atk")  -> attacker (guess + one fault)
//   agent "env" = input("env")  -> secret PIN (read after the attacker)
//
// Single fault is insufficient -> bypass not robust:
//   -atl "<atk>F{auth == 1}"     (FALSE = SAFE, desirable outcome)
// =====================================================================

int main() {
    int guess = input("atk");    // attacker's entered PIN
    int fault = input("atk");    // attacker injects one fault
    int secret = input("env");   // secret PIN (read after the attacker commits)
    int diff = 0;
    int diff2 = 0;
    if (guess != secret) { diff = 1; }
    if (guess != secret) { diff2 = 1; }   // redundant comparison
    // a single fault can only hit one of the two checks
    if (fault == 1) { diff = 0; }
    int auth = 0;
    if (diff == 0) {
        if (diff2 == 0) { auth = 1; }     // both must agree
    }
    while (1) {}
}
